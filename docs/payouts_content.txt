                let vendorPayoutMethods = [];

                async function renderVendorPayouts() {
                    const loader = document.getElementById('payoutMethodsLoader');
                    const errorEl = document.getElementById('payoutMethodsError');
                    const emptyEl = document.getElementById('payoutMethodsEmpty');
                    const listEl = document.getElementById('payoutMethodsList');
                    const escrowSelect = document.getElementById('payoutEscrowTimerSelect');

                    loader.classList.remove('hidden');
                    errorEl.classList.add('hidden');
                    emptyEl.classList.add('hidden');
                    listEl.classList.add('hidden');

                    try {
                        // Load current user profile for escrow timer
                        const { data: { user } } = await window.supabaseClient.auth.getUser();
                        if (user && user.user_metadata && user.user_metadata.escrow_timer_hours) {
                            escrowSelect.value = user.user_metadata.escrow_timer_hours;
                        }

                        // Load payout methods via API
                        const { data: session } = await window.supabaseClient.auth.getSession();
                        const token = session?.session?.access_token;
                        if (!token) throw new Error("Non authentifié");

                        const response = await fetch('/api/vendor-payouts/methods', {
                            headers: { 'Authorization': `Bearer ${token}` }
                        });

                        const result = await response.json();
                        
                        if (!response.ok) {
                            throw new Error(result.message || "Erreur lors du chargement des comptes.");
                        }

                        vendorPayoutMethods = result.data || [];

                        if (vendorPayoutMethods.length === 0) {
                            emptyEl.classList.remove('hidden');
                        } else {
                            renderPayoutMethodsList(vendorPayoutMethods, listEl);
                            listEl.classList.remove('hidden');
                        }
                    } catch (error) {
                        console.error("Erreur renderVendorPayouts:", error);
                        errorEl.textContent = error.message || "Impossible de charger vos comptes. Veuillez réessayer.";
                        errorEl.classList.remove('hidden');
                    } finally {
                        loader.classList.add('hidden');
                    }
                }
                window.renderVendorPayouts = renderVendorPayouts;

                function renderPayoutMethodsList(methods, container) {
                    container.innerHTML = '';
                    methods.forEach(method => {
                        const isDefault = method.is_default;
                        
                        let providerIcon = 'solar:wallet-money-bold';
                        let providerColor = 'text-gray-500';
                        if(method.provider === 'STRIPE') { providerIcon = 'logos:stripe'; providerColor = ''; }
                        else if(method.provider === 'MOBILE_MONEY') { providerIcon = 'solar:phone-bold'; providerColor = 'text-green-600'; }
                        else if(method.provider === 'BANK_TRANSFER') { providerIcon = 'solar:bank-bold'; providerColor = 'text-blue-600'; }

                        const card = document.createElement('div');
                        card.className = `border rounded-xl p-4 flex flex-col sm:flex-row sm:items-center justify-between gap-4 transition-all ${isDefault ? 'border-[#013220] bg-[#013220]/5' : 'border-gray-200 hover:border-gray-300'}`;
                        
                        card.innerHTML = `
                            <div class="flex items-center gap-4">
                                <div class="w-12 h-12 rounded-xl bg-white border border-gray-100 flex items-center justify-center shrink-0 shadow-sm">
                                    <iconify-icon icon="${providerIcon}" width="24" class="${providerColor}"></iconify-icon>
                                </div>
                                <div>
                                    <div class="flex items-center gap-2">
                                        <h4 class="font-bold text-gray-900">${method.provider}</h4>
                                        ${isDefault ? '<span class="bg-[#013220] text-white text-[10px] uppercase tracking-wider font-bold px-2 py-0.5 rounded">Principal</span>' : ''}
                                    </div>
                                    <p class="text-sm text-gray-500 font-medium font-mono mt-0.5">**** ${method.account_id.slice(-4)}</p>
                                    <p class="text-xs text-gray-400 mt-1">Titulaire : ${method.account_name}</p>
                                </div>
                            </div>
                            <div class="flex items-center gap-2">
                                <button onclick="deletePayoutMethod('${method.id}')" class="p-2 text-gray-400 hover:text-red-600 hover:bg-red-50 rounded-lg transition-colors" title="Supprimer ce compte">
                                    <iconify-icon icon="solar:trash-bin-trash-bold" width="20"></iconify-icon>
                                </button>
                            </div>
                        `;
                        container.appendChild(card);
                    });
                }

                async function updateEscrowTimer(hours) {
                    const msgEl = document.getElementById('escrowTimerUpdateMsg');
                    msgEl.classList.add('hidden');
                    
                    try {
                        const { data: session } = await window.supabaseClient.auth.getSession();
                        const token = session?.session?.access_token;
                        if (!token) return;

                        const response = await fetch('/api/vendor-payouts/escrow-timer', {
                            method: 'PUT',
                            headers: { 
                                'Authorization': `Bearer ${token}`,
                                'Content-Type': 'application/json'
                            },
                            body: JSON.stringify({ hours: parseInt(hours) })
                        });

                        if (response.ok) {
                            msgEl.textContent = "Mis à jour !";
                            msgEl.className = "text-sm text-green-600 font-medium";
                            msgEl.classList.remove('hidden');
                            setTimeout(() => msgEl.classList.add('hidden'), 3000);
                        } else {
                            throw new Error("Erreur");
                        }
                    } catch (error) {
                        console.error("Erreur updateEscrowTimer:", error);
                        msgEl.textContent = "Erreur de mise à jour";
                        msgEl.className = "text-sm text-red-600 font-medium";
                        msgEl.classList.remove('hidden');
                    }
                }
                window.updateEscrowTimer = updateEscrowTimer;

                async function requestPayoutOtp() {
                    const provider = document.getElementById('payoutProvider').value;
                    const accountId = document.getElementById('payoutAccountId').value.trim();
                    const accountName = document.getElementById('payoutAccountName').value.trim();
                    const errorEl = document.getElementById('payoutFormError');
                    const btn = document.getElementById('payoutRequestOtpBtn');

                    if(!accountId || !accountName) {
                        errorEl.textContent = "Veuillez remplir tous les champs.";
                        errorEl.classList.remove('hidden');
                        return;
                    }

                    errorEl.classList.add('hidden');
                    btn.disabled = true;
                    const originalText = btn.innerHTML;
                    btn.innerHTML = '<iconify-icon icon="line-md:loading-twotone-loop" width="20"></iconify-icon> Envoi...';

                    try {
                        const { data: session } = await window.supabaseClient.auth.getSession();
                        const token = session?.session?.access_token;
                        if (!token) throw new Error("Non authentifié");

                        const response = await fetch('/api/vendor-payouts/request-otp', {
                            method: 'POST',
                            headers: { 
                                'Authorization': `Bearer ${token}`,
                                'Content-Type': 'application/json'
                            },
                            body: JSON.stringify({ action: 'add_payout_method' })
                        });

                        const result = await response.json();
                        
                        if (!response.ok) {
                            throw new Error(result.message || "Erreur lors de l'envoi de l'OTP.");
                        }

                        currentPayoutOtpToken = result.data.token;
                        
                        // Show OTP Step
                        document.getElementById('payoutStepForm').classList.add('hidden');
                        document.getElementById('payoutStepOtp').classList.remove('hidden');
                        
                    } catch (error) {
                        errorEl.textContent = error.message;
                        errorEl.classList.remove('hidden');
                    } finally {
                        btn.disabled = false;
                        btn.innerHTML = originalText;
                    }
                }
                window.requestPayoutOtp = requestPayoutOtp;

                async function verifyPayoutOtpAndAdd() {
                    const otpCode = document.getElementById('payoutOtpInput').value.trim();
                    const provider = document.getElementById('payoutProvider').value;
                    const accountId = document.getElementById('payoutAccountId').value.trim();
                    const accountName = document.getElementById('payoutAccountName').value.trim();
                    const errorEl = document.getElementById('payoutOtpError');
                    const btn = document.getElementById('payoutVerifyOtpBtn');

                    if(!otpCode || otpCode.length !== 6 || !currentPayoutOtpToken) {
                        errorEl.textContent = "Veuillez entrer un code valide à 6 chiffres.";
                        errorEl.classList.remove('hidden');
                        return;
                    }

                    errorEl.classList.add('hidden');
                    btn.disabled = true;
                    const originalText = btn.innerHTML;
                    btn.innerHTML = '<iconify-icon icon="line-md:loading-twotone-loop" width="20"></iconify-icon> Vérification...';

                    try {
                        const { data: session } = await window.supabaseClient.auth.getSession();
                        const token = session?.session?.access_token;
                        if (!token) throw new Error("Non authentifié");

                        const response = await fetch('/api/vendor-payouts/add', {
                            method: 'POST',
                            headers: { 
                                'Authorization': `Bearer ${token}`,
                                'Content-Type': 'application/json'
                            },
                            body: JSON.stringify({ 
                                provider,
                                account_id: accountId,
                                account_name: accountName,
                                token: currentPayoutOtpToken,
                                code: otpCode
                            })
                        });

                        const result = await response.json();
                        
                        if (!response.ok) {
                            throw new Error(result.message || "Code invalide ou expiré.");
                        }

                        // Success
                        closeAddPayoutModal();
                        renderVendorPayouts(); // Refresh list
                        
                        // Show toast if ToastMaker exists
                        if(window.ToastMaker) window.ToastMaker.success("Compte de réception ajouté avec succès !");

                    } catch (error) {
                        errorEl.textContent = error.message;
                        errorEl.classList.remove('hidden');
                    } finally {
                        btn.disabled = false;
                        btn.innerHTML = originalText;
                    }
                }
                window.verifyPayoutOtpAndAdd = verifyPayoutOtpAndAdd;

                async function deletePayoutMethod(methodId) {
                    if(!confirm("Êtes-vous sûr de vouloir supprimer ce compte de réception ?")) return;
                    
                    try {
                        const { data: session } = await window.supabaseClient.auth.getSession();
                        const token = session?.session?.access_token;
                        if (!token) return;

                        // Here we assume backend DELETE route might need OTP in future, but let's try direct call first.
                        const response = await fetch(`/api/vendor-payouts/methods/${methodId}`, {
                            method: 'DELETE',
                            headers: { 
                                'Authorization': `Bearer ${token}`,
                                'Content-Type': 'application/json'
                            },
                            body: JSON.stringify({}) // Add token/code here if backend requires OTP for delete
                        });

                        const result = await response.json();
                        if (!response.ok) {
                            // If it requires OTP, we would show an OTP modal here.
                            throw new Error(result.message || "Erreur lors de la suppression.");
                        }

                        if(window.ToastMaker) window.ToastMaker.success("Compte supprimé.");
                        renderVendorPayouts();

                    } catch(e) {
                        alert(e.message);
                    }
                }
